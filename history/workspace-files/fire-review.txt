/** Vanilla fire geometry/sounds/hooks, with a faster bounded spread and independent magic damage. */
public final class MagicFireBlock extends FireBlock {
 public static final MapCodec<MagicFireBlock> CODEC=simpleCodec(MagicFireBlock::new);
 public static final IntegerProperty PULSE=IntegerProperty.create("pulse",0,3);
 public MagicFireBlock(Properties p){super(p);registerDefaultState(defaultBlockState().setValue(PULSE,0));}
 @SuppressWarnings({"unchecked","rawtypes"})
 @Override public MapCodec<FireBlock> codec(){return (MapCodec)CODEC;}
 @Override protected void createBlockStateDefinition(StateDefinition.Builder<Block,BlockState> b){super.createBlockStateDefinition(b);b.add(PULSE);}
 public BlockState stateAt(BlockGetter l,BlockPos p){
  var below=p.below();var bs=l.getBlockState(below);var s=defaultBlockState();
  if(bs.isFaceSturdy(l,below,Direction.UP) || bs.isFlammable(l,below,Direction.UP))return s;
  return s.setValue(NORTH,flammable(l,p.north(),Direction.SOUTH)).setValue(SOUTH,flammable(l,p.south(),Direction.NORTH)).setValue(EAST,flammable(l,p.east(),Direction.WEST)).setValue(WEST,flammable(l,p.west(),Direction.EAST)).setValue(UP,flammable(l,p.above(),Direction.DOWN));
 }
 private boolean flammable(BlockGetter l,BlockPos p,Direction f){return l.getBlockState(p).isFlammable(l,p,f);}
 @Override public BlockState getStateForPlacement(net.minecraft.world.item.context.BlockPlaceContext c){return stateAt(c.getLevel(),c.getClickedPos());}
 @Override protected BlockState getStateForPlacement(BlockGetter l,BlockPos p){return stateAt(l,p);}
 @Override protected BlockState updateShape(BlockState s,Direction d,BlockState neighbor,LevelAccessor l,BlockPos p,BlockPos other){return canSurvive(s,l,p)?stateAt(l,p).setValue(AGE,s.getValue(AGE)).setValue(PULSE,s.getValue(PULSE)):Blocks.AIR.defaultBlockState();}
 private static int delay(int step){int total=Config.FIRE_SECONDS.get()*60;return Math.max(1,total/64+(step<total%64?1:0));}
 @Override protected void onPlace(BlockState s,Level l,BlockPos p,BlockState old,boolean moving){if(!old.is(this)){l.scheduleTick(p,this,delay(s.getValue(AGE)*4+s.getValue(PULSE)));}}
 public static boolean lightCampfire(Level l,BlockPos p){var s=l.getBlockState(p);if(CampfireBlock.canLight(s)){l.setBlock(p,s.setValue(CampfireBlock.LIT,true),3);l.playSound(null,p,net.minecraft.sounds.SoundEvents.FLINTANDSTEEL_USE,net.minecraft.sounds.SoundSource.BLOCKS,1,1);return true;}return false;}
 @Override protected void tick(BlockState s,ServerLevel l,BlockPos p,RandomSource r){
  int step=s.getValue(AGE)*4+s.getValue(PULSE);
  if(!l.getGameRules().getBoolean(GameRules.RULE_DOFIRETICK)){l.scheduleTick(p,this,delay(step));return;}
  if(!canSurvive(s,l,p)){l.removeBlock(p,false);return;}
  boolean eternal=l.getBlockState(p.below()).isFireSource(l,p.below(),Direction.UP);
  if(!eternal && l.isRainingAt(p)){l.removeBlock(p,false);return;}
  // Six neighboring blocks only: expected burn rate > vanilla (flammability / 70, every ~10 ticks).
  for(Direction d:Direction.values()){
   BlockPos q=p.relative(d);if(!l.hasChunkAt(q)||!l.getWorldBorder().isWithinBounds(q))continue;
   if(lightCampfire(l,q))continue;
   var target=l.getBlockState(q);int odds=target.getFlammability(l,q,d.getOpposite());
   if(odds>0 && r.nextInt(70)<odds){
    target.onCaughtFire(l,q,d.getOpposite(),null);l.removeBlock(q,false);
    var next=stateAt(l,q);if(next.canSurvive(l,q) && !l.isRainingAt(q))l.setBlock(q,next,3);
   }
  }
  // Bounded local spread, preserving purple fire; honors modded flammability and doFireTick.
  for(int dx=-1;dx<=1;dx++)for(int dz=-1;dz<=1;dz++)for(int dy=0;dy<=1;dy++){
   if(dx==0&&dz==0&&dy==0)continue;BlockPos q=p.offset(dx,dy,dz);
   if(!l.hasChunkAt(q)||!l.getWorldBorder().isWithinBounds(q)||!l.isEmptyBlock(q)||l.isRainingAt(q))continue;
   int odds=0;for(Direction d:Direction.values())odds=Math.max(odds,l.getBlockState(q.relative(d)).getFireSpreadSpeed(l,q.relative(d),d.getOpposite()));
   var next=stateAt(l,q);if(odds>0 && r.nextInt(100)<odds && next.canSurvive(l,q))l.setBlock(q,next,3);
  }
  // Neighbor burns may remove the current block's support.
  if(!l.getBlockState(p).is(this))return;
  if(!eternal && step>=63){l.removeBlock(p,false);return;}
  step=eternal?0:step+1;l.setBlock(p,stateAt(l,p).setValue(AGE,step/4).setValue(PULSE,step%4),2);l.scheduleTick(p,this,delay(step));
 }
 @Override protected void entityInside(BlockState s,Level l,BlockPos p,Entity e){
  if(l.isClientSide)return;
  if(e instanceof LivingEntity live)ArcaneFire.ignite(live);
  else if(e instanceof ItemEntity item && !item.fireImmune())item.hurt(l.damageSources().inFire(),2);
 }
 // Inherits vanilla fire ambience and smoke. No following dust particles on entities.
}
package dev.kelmagic.server;
import dev.kelmagic.*;
import dev.kelmagic.network.BurnSnapshot;
import net.minecraft.core.registries.Registries;
import net.minecraft.resources.*;
import net.minecraft.world.damagesource.*;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.player.Player;
import net.minecraft.server.level.ServerPlayer;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.tick.EntityTickEvent;
import net.neoforged.neoforge.event.entity.player.PlayerEvent;
import net.neoforged.neoforge.network.PacketDistributor;
@EventBusSubscriber(modid=KelMagic.MODID)
public final class ArcaneFire {
 public static final ResourceKey<DamageType> DAMAGE=ResourceKey.create(Registries.DAMAGE_TYPE,ResourceLocation.fromNamespaceAndPath(KelMagic.MODID,"arcane_fire"));
 private static void sync(LivingEntity e){PacketDistributor.sendToPlayersTrackingEntityAndSelf(e,new BurnSnapshot(e.getId(),e.getData(MagicAttachments.ARCANE_BURN)));}
 public static boolean ignite(LivingEntity e){
  if(!e.isAlive() || e.isSpectator() || e instanceof Player p && p.getAbilities().invulnerable)return false;
  int old=e.getData(MagicAttachments.ARCANE_BURN);e.setData(MagicAttachments.ARCANE_BURN,Math.max(old,Config.FIRE_SECONDS.get()*60));if(old==0)sync(e);return true;
 }
 public static void advance(LivingEntity e){
  int left=e.getData(MagicAttachments.ARCANE_BURN);if(left<=0)return;
  if(!e.isAlive() || e.isSpectator() || e instanceof Player p && p.getAbilities().invulnerable || e.isInWaterRainOrBubble()){e.setData(MagicAttachments.ARCANE_BURN,0);sync(e);return;}
  e.setData(MagicAttachments.ARCANE_BURN,left-1);
  if(e.tickCount%20==0){
   var velocity=e.getDeltaMovement();boolean impulse=e.hasImpulse;
   e.hurt(new DamageSource(e.level().registryAccess().registryOrThrow(Registries.DAMAGE_TYPE).getHolderOrThrow(DAMAGE)),1);
   e.setDeltaMovement(velocity);e.hasImpulse=impulse;
  }
  if(left==1 || e.tickCount%20==0)sync(e);
 }
 @SubscribeEvent public static void tick(EntityTickEvent.Post e){if(!e.getEntity().level().isClientSide && e.getEntity() instanceof LivingEntity l)advance(l);}
 @SubscribeEvent public static void tracking(PlayerEvent.StartTracking e){if(e.getEntity() instanceof ServerPlayer observer && e.getTarget() instanceof LivingEntity target)PacketDistributor.sendToPlayer(observer,new BurnSnapshot(target.getId(),target.getData(MagicAttachments.ARCANE_BURN)));}
 @SubscribeEvent public static void login(PlayerEvent.PlayerLoggedInEvent e){if(e.getEntity() instanceof ServerPlayer p)sync(p);}
 @SubscribeEvent public static void dimension(PlayerEvent.PlayerChangedDimensionEvent e){if(e.getEntity() instanceof ServerPlayer p)sync(p);}
 private ArcaneFire(){}
}
package dev.kelmagic.server;

import dev.kelmagic.Config;
import dev.kelmagic.core.SpellParser.Effect;
import java.util.*;
import net.minecraft.core.BlockPos;
import net.minecraft.core.particles.ParticleTypes;
import net.minecraft.network.chat.Component;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.effect.*;
import net.minecraft.world.entity.*;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.level.ClipContext;
import net.minecraft.world.level.block.BaseFireBlock;
import net.minecraft.world.phys.*;

public final class SpellEffects {
    public record Aim(Vec3 point, HitResult.Type type, BlockHitResult block, LivingEntity entity) {}
    static Aim aim(ServerPlayer p) {
        Vec3 eye=p.getEyePosition(), end=eye.add(p.getLookAngle().scale(Config.RANGE.get()));
        BlockHitResult block=p.serverLevel().clip(new ClipContext(eye,end,ClipContext.Block.COLLIDER,ClipContext.Fluid.NONE,p));
        Vec3 stop=block.getLocation(); LivingEntity target=null; double distance=eye.distanceToSqr(stop);
        AABB box=new AABB(eye,stop).inflate(1);
        for(LivingEntity e:p.serverLevel().getEntitiesOfClass(LivingEntity.class,box,e -> e!=p && e.isAlive() && !e.isSpectator())) {
            Optional<Vec3> hit=e.getBoundingBox().inflate(0.15).clip(eye,stop);
            if(hit.isPresent() && eye.distanceToSqr(hit.get())<distance) { distance=eye.distanceToSqr(hit.get()); target=e; }
        }
        return new Aim(target==null?stop:target.position(),target==null?block.getType():HitResult.Type.ENTITY,block,target);
    }
    public static boolean execute(ServerPlayer p,Effect effect) {
        Aim aim=aim(p);
        return switch(effect) {
            case FIRE_SELF -> { ignite(p); yield true; }
            case RESIST_SELF -> { p.addEffect(new MobEffectInstance(MobEffects.FIRE_RESISTANCE,Config.RESIST_SECONDS.get()*20)); yield true; }
            case FIRE_PLAYER -> aim.entity instanceof Player other && canHarm(p,other) && ignite(other);
            case FIRE_MOB -> aim.entity instanceof Mob mob && ignite(mob);
            case FIRE_BLOCK -> aim.type==HitResult.Type.BLOCK && placeFire(p,aim.block.getBlockPos().relative(aim.block.getDirection()));
            case FIRE_AREA -> areaFire(p,aim);
            case FIRE_PLAYERS -> areaTargets(p,aim,true);
            case FIRE_MOBS -> areaTargets(p,aim,false);
            default -> false;
        };
    }
    private static boolean canHarm(ServerPlayer caster,Player target) {
        return target==caster || caster.server.isPvpAllowed() && caster.canHarmPlayer(target);
    }
    private static boolean ignite(LivingEntity target) {
        return ArcaneFire.ignite(target);
    }
    private static boolean areaTargets(ServerPlayer p,Aim aim,boolean players) {
        double radius=Config.AREA_RADIUS.get();
        var entities=p.serverLevel().getEntitiesOfClass(LivingEntity.class,new AABB(aim.point,aim.point).inflate(radius),
            e -> e.isAlive() && !e.isSpectator() && e.distanceToSqr(aim.point)<=radius*radius && (players?e instanceof Player:e instanceof Mob));
        entities.sort(Comparator.<LivingEntity>comparingDouble(e -> e.distanceToSqr(aim.point)).thenComparing(e -> e.getUUID().toString()));
        int hits=0;
        for(LivingEntity e:entities) {
            if(hits==Config.MAX_TARGETS.get()) break;
            if(e instanceof Player player && !canHarm(p,player)) continue;
            // Neither the target ray nor a lu expansion should pass through solid walls.
            var obstruction=p.serverLevel().clip(new ClipContext(aim.point.add(0,0.5,0),e.getEyePosition(),ClipContext.Block.COLLIDER,ClipContext.Fluid.NONE,p));
            if(obstruction.getType()!=HitResult.Type.MISS && obstruction.getLocation().distanceToSqr(e.getEyePosition())>1) continue;
            if(ignite(e)) hits++;
        }
        return hits>0;
    }
    private static boolean placeFire(ServerPlayer p,BlockPos pos) {
        var level=p.serverLevel();
        if(!level.hasChunkAt(pos) || !level.getWorldBorder().isWithinBounds(pos) || !level.isInWorldBounds(pos)
                || !level.mayInteract(p,pos) || !p.mayUseItemAt(pos,net.minecraft.core.Direction.UP,p.getMainHandItem()) || !level.isEmptyBlock(pos)) return false;
        var state=dev.kelmagic.MagicContent.MAGIC_FIRE.get().stateAt(level,pos);
        if(!state.canSurvive(level,pos)) return false;
        var snapshot=net.neoforged.neoforge.common.util.BlockSnapshot.create(level.dimension(),level,pos);
        if(!level.setBlockAndUpdate(pos,state)) return false;
        if(net.neoforged.neoforge.event.EventHooks.onBlockPlace(p,snapshot,net.minecraft.core.Direction.UP)) {
            snapshot.restore(); return false;
        }
        return true;
    }
    private static boolean areaFire(ServerPlayer p,Aim aim) {
        if(aim.type!=HitResult.Type.BLOCK) return false;
        int r=Config.AREA_RADIUS.get(),hits=0; BlockPos origin=aim.block.getBlockPos().relative(aim.block.getDirection());
        List<BlockPos> positions=new ArrayList<>();
        for(int x=-r;x<=r;x++) for(int z=-r;z<=r;z++) if(x*x+z*z<=r*r) positions.add(origin.offset(x,0,z));
        positions.sort(Comparator.<BlockPos>comparingDouble(pos -> pos.distSqr(origin)).thenComparingInt(BlockPos::getX).thenComparingInt(BlockPos::getZ));
        for(BlockPos pos:positions) {
            if(hits>=Config.MAX_FIRES.get()) break;
            for(int y=2;y>=-2;y--) if(placeFire(p,pos.above(y))) { hits++; break; }
        }
        return hits>0;
    }
    public static void fizzle(ServerPlayer p,String key) {
        p.displayClientMessage(Component.translatable(key),true);
        if(key.equals("kelmagic.undefined")) {
            // Real explosive firework attached to the caster: native damage, creative/PvP rules.
            var rocket=new net.minecraft.world.item.ItemStack(net.minecraft.world.item.Items.FIREWORK_ROCKET);
            var star=new net.minecraft.world.item.component.FireworkExplosion(
                net.minecraft.world.item.component.FireworkExplosion.Shape.STAR,
                it.unimi.dsi.fastutil.ints.IntList.of(0x67E6ED,0xB982E6),
                it.unimi.dsi.fastutil.ints.IntList.of(0xEBC478),true,true);
            rocket.set(net.minecraft.core.component.DataComponents.FIREWORKS,
                new net.minecraft.world.item.component.Fireworks(0,java.util.List.of(star)));
            p.serverLevel().addFreshEntity(new net.minecraft.world.entity.projectile.FireworkRocketEntity(p.serverLevel(),rocket,p));
        } else p.serverLevel().sendParticles(ParticleTypes.FIREWORK,p.getX(),p.getY()+2.5,p.getZ(),24,0.4,0.3,0.4,0.06);
    }
    private SpellEffects() {}
}
